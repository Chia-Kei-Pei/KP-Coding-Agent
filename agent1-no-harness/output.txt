This is a classic project for learning game development. We will use the `pygame` library in Python, as it provides excellent tools for handling graphics, input, and game loops.

Since the game requires external libraries, remember to install `pygame` first:
```bash
pip install pygame
```

Here is the code in the `src` folder.

### `src/pong.py`

```python
import pygame
import sys

# --- Initialization ---
pygame.init()

# --- Constants ---
SCREEN_WIDTH = 800
SCREEN_HEIGHT = 600
FPS = 60

# Colors (RGB)
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)

# Setup Screen
screen = pygame.display.set_mode((SCREEN_WIDTH, SCREEN_HEIGHT))
pygame.display.set_caption("Pygame Pong")
clock = pygame.time.Clock()

# --- Game Objects Configuration ---
PADDLE_WIDTH = 15
PADDLE_HEIGHT = 100
BALL_SIZE = 15

# --- Classes ---

class Paddle(pygame.sprite.Sprite):
    """Represents a player-controlled paddle."""
    def __init__(self, x, y, color):
        super().__init__()
        self.image = pygame.Surface([PADDLE_WIDTH, PADDLE_HEIGHT])
        self.image.fill(color)
        self.rect = self.image.get_rect()
        self.rect.x = x
        self.rect.y = y
        self.speed = 8
        self.color = color

    def move(self, direction):
        """Moves the paddle by a given direction (-1 for up, 1 for down)."""
        self.rect.y += direction * self.speed
        
        # Keep paddle within screen boundaries
        if self.rect.top < 0:
            self.rect.top = 0
        if self.rect.bottom > SCREEN_HEIGHT:
            self.rect.bottom = SCREEN_HEIGHT

class Ball(pygame.sprite.Sprite):
    """Represents the moving ball."""
    def __init__(self):
        super().__init__()
        self.image = pygame.Surface([BALL_SIZE, BALL_SIZE])
        self.image.fill(WHITE)
        self.rect = self.image.get_rect()
        
        # Starting position (center)
        self.rect.x = SCREEN_WIDTH // 2 - BALL_SIZE // 2
        self.rect.y = SCREEN_HEIGHT // 2 - BALL_SIZE // 2
        
        # Initial speed/direction (randomized slightly)
        self.speed_x = 5
        self.speed_y = 5

    def update(self):
        """Moves the ball and handles wall collisions."""
        self.rect.x += self.speed_x
        self.rect.y += self.speed_y

        # Collision with top and bottom walls
        if self.rect.top <= 0 or self.rect.bottom >= SCREEN_HEIGHT:
            self.speed_y *= -1
        
    def reset(self, scoring_player):
        """Resets the ball to the center after a point is scored."""
        self.rect.x = SCREEN_WIDTH // 2 - BALL_SIZE // 2
        self.rect.y = SCREEN_HEIGHT // 2 - BALL_SIZE // 2
        
        # Serve the ball toward the player who *did not* score
        self.speed_x = -5 if scoring_player == 1 else 5
        self.speed_y = 5

# --- Game Logic Functions ---

def handle_collisions(ball, player1, player2):
    """Checks collision with paddles and reverses ball direction."""
    
    # Collision with Player 1 (Left Paddle)
    if ball.rect.colliderect(player1.rect):
        # Ensure the ball is moving towards the paddle
        if ball.speed_x < 0:
            ball.speed_x *= -1.05  # Reverse and slightly increase speed
            # Simple angle deflection based on where the hit occurred
            offset = (player1.rect.centery - ball.rect.centery) / (PADDLE_HEIGHT / 2)
            ball.speed_y = offset * 6

    # Collision with Player 2 (Right Paddle)
    if ball.rect.colliderect(player2.rect):
        if ball.speed_x > 0:
            ball.speed_x *= -1.05
            offset = (player2.rect.centery - ball.rect.centery) / (PADDLE_HEIGHT / 2)
            ball.speed_y = offset * 6

def check_scoring(ball, score_p1, score_p2):
    """Checks if the ball passed the left or right boundaries."""
    
    # Scoring for Player 2 (Left side goal)
    if ball.rect.left <= 0:
        score_p2[0] += 1
        ball.reset(scoring_player=2) # Player 2 scored, ball serves right
        return True

    # Scoring for Player 1 (Right side goal)
    if ball.rect.right >= SCREEN_WIDTH:
        score_p1[0] += 1
        ball.reset(scoring_player=1) # Player 1 scored, ball serves left
        return True
        
    return False

def draw_elements(screen, player1, player2, ball, score_p1, score_p2):
    """Draws all game elements and the score."""
    
    # Background and dividing line
    screen.fill(BLACK)
    pygame.draw.aaline(screen, WHITE, pygame.Rect(SCREEN_WIDTH // 2 - 2, 0, 4, SCREEN_HEIGHT))
    
    # Draw sprites
    all_sprites = pygame.sprite.Group(player1, player2, ball)
    all_sprites.draw(screen)
    
    # Draw Score
    font = pygame.font.Font(None, 74)
    p1_text = font.render(f"{score_p1[0]}", True, WHITE)
    p2_text = font.render(f"{score_p2[0]}", True, WHITE)
    
    screen.blit(p1_text, (SCREEN_WIDTH // 4, 20))
    screen.blit(p2_text, (SCREEN_WIDTH * 3 // 4 - p2_text.get_width(), 20))


# --- Main Game Loop ---

def main():
    # Initialize scores (using lists so they can be passed and modified)
    score_p1 = [0]
    score_p2 = [0]
    
    # Create game objects
    player1 = Paddle(20, SCREEN_HEIGHT // 2 - PADDLE_HEIGHT // 2, (0, 255, 0)) # Green
    player2 = Paddle(SCREEN_WIDTH - 20 - PADDLE_WIDTH, SCREEN_HEIGHT // 2 - PADDLE_HEIGHT // 2, (255, 0, 0)) # Red
    ball = Ball()
    
    # Setup sprite groups for collision detection
    all_sprites = pygame.sprite.Group(player1, player2, ball)

    running = True
    while running:
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                running = False
                pygame.quit()
                sys.exit()

        # 1. Player Input Handling (Two-Player Setup)
        keys = pygame.key.get_pressed()
        
        # Player 1 Controls (W/S)
        if keys[pygame.K_w]:
            player1.move(-1) # Move up
        if keys[pygame.K_s]:
            player1.move(1) # Move down

        # Player 2 Controls (UP/DOWN)
        if keys[pygame.K_UP]:
            player2.move(-1) # Move up
        if keys[pygame.K_DOWN]:
            player2.move(1) # Move down

        # 2. Update Game State
        all_sprites.update()
        
        # 3. Collision Handling (Paddle <-> Ball)
        handle_collisions(ball, player1, player2)
        
        # 4. Scoring and Boundary Check (Ball <-> Walls/Goals)
        # Returns True if a score occurred, forcing a ball reset
        score_p1_scored = check_scoring(ball, score_p1, score_p2)
        score_p2_scored = check_scoring(ball, score_p1, score_p2)

        # 5. Drawing
        draw_elements(screen, player1, player2, ball, score_p1, score_p2)
        
        # Update display and cap framerate
        pygame.display.flip()
        clock.tick(FPS)

if __name__ == '__main__':
    main()

```

### How to Run the Game

1.  Save the code above as `pong.py` inside your `src` folder.
2.  Open your terminal or command prompt and navigate to the `src` folder.
3.  Run the game:
    ```bash
    python pong.py
    ```

### Game Controls

This version is set up for a two-player experience:

*   **Player 1 (Left - Green):**
    *   Move Up: **W**
    *   Move Down: **S**
*   **Player 2 (Right - Red):**
    *   Move Up: **↑** (Up Arrow)
    *   Move Down: **↓** (Down Arrow)

### Code Explanation

1.  **Structure:** We use `pygame.sprite.Sprite` to define `Paddle` and `Ball`. This is the standard object-oriented way to manage game entities in Pygame, making drawing and updating efficient.
2.  **`Paddle.move()`:** This method handles player input by adjusting the `rect.y` property, while including boundary checks to keep the paddle on the screen.
3.  **`Ball.update()`:** This is the physics engine for the ball. It changes the `rect.x` and `rect.y` based on `speed_x` and `speed_y`. It also reverses `speed_y` whenever it hits the top or bottom boundary.
4.  **`handle_collisions()`:** This is the critical collision logic. When the ball hits a paddle (`colliderect`), it does two things:
    *   It reverses the horizontal direction (`speed_x *= -1.05`).
    *   It calculates a deflection angle based on *where* the paddle was hit, giving the ball a more dynamic bounce (hitting the top edge makes it go up faster, hitting the middle keeps the speed similar).
5.  **`check_scoring()`:** This checks if the ball has moved past the left or right edge of the screen. If it has, a point is awarded, and `ball.reset()` is called to place the ball back in the center, serving toward the opponent.
6.  **`main()` Loop:** This loop is the heart of the game. It continuously:
    *   Handles **Events** (like quitting the game).
    *   Reads **Input** (checking which keys are held down to move the paddles).
    *   Calls the **Update** methods (movement, physics, collision checks).
    *   Calls the **Drawing** function (clearing the screen and redrawing all elements).
    *   Uses `clock.tick(FPS)` to ensure the game runs at a consistent speed (60 frames per second).
